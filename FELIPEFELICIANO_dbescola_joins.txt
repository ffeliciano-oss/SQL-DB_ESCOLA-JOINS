-- Criando a tabela Matriculas
CREATE TABLE Matriculas (
    cod INT PRIMARY KEY,
    valor DECIMAL(10, 2),
    data DATE,
    cod_aluno INT,
    FOREIGN KEY (cod_aluno) REFERENCES Alunos(cod)
);

/*2. Alunos matriculados
Liste todos os alunos que possuem matrículas, mostrando o nome do aluno, a cidade onde mora e o valor da matrícula.*/
select
	Alu.cod as cod_alu,
    Alu.nome as nome_alu,
    Alu.cidade as cidade_alu,
    Mat.cod as cod_mat,
    Mat.valor as valor_mat
from Alunos as Alu
inner join Matriculas Mat
on Alu.cod = Mat.cod_aluno
order by Alu.cod,Mat.cod;

/*3. Todos os alunos e suas matrículas
Liste todos os alunos, mostrando o nome e o valor da matrícula, mesmo que alguns alunos não tenham realizado nenhuma matrícula.*/
select
	Alu.cod as cod_alu,
    Alu.nome as nome_alu,
    Alu.cidade as cidade_alu,
    Mat.cod as cod_mat,
    Mat.valor as valor_mat
from Alunos as Alu
left join Matriculas Mat
on Alu.cod = Mat.cod_aluno
order by Alu.cod,Mat.cod;

/*4. Matrículas e alunos correspondentes
Liste todas as matrículas, incluindo os dados dos alunos correspondentes, mesmo que existam alunos sem nenhuma matrícula.*/
select
	Alu.cod as cod_alu,
    Alu.nome as nome_alu,
    Alu.cidade as cidade_alu,
    Mat.cod as cod_mat,
    Mat.valor as valor_mat
from Alunos as Alu
right join Matriculas Mat
on Alu.cod = Mat.cod_aluno
order by Alu.cod,Mat.cod;

/*5. Alunos sem matrícula
Liste todos os alunos que ainda não realizaram nenhuma matrícula.*/
select
	Alu.cod as cod_alu,
    Alu.nome as nome_alu,
    Alu.cidade as cidade_alu,
    Mat.cod as cod_mat,
    Mat.valor as valor_mat
from Alunos as Alu
left join Matriculas Mat
on Alu.cod = Mat.cod_aluno
where Mat.cod is null
order by Alu.cod,Mat.cod;

/*6. Matrículas acima de determinado valor
Liste todas as matrículas com valor acima de R$ 200,00, mostrando o nome do aluno e o valor da matrícula.*/
select
	Alu.cod as cod_alu,
    Alu.nome as nome_alu,
    Alu.cidade as cidade_alu,
    Mat.cod as cod_mat,
    Mat.valor as valor_mat
from Alunos as Alu
right join Matriculas Mat
on Alu.cod = Mat.cod_aluno
where Mat.valor > 200.00
order by Alu.cod,Mat.cod;

/*7. Matrículas de outubro de 2023
Liste todas as matrículas realizadas no mês de outubro de 2023, mostrando o nome do aluno, a data e o valor da matrícula.*/
select
	Alu.cod as cod_alu,
    Alu.nome as nome_alu,
    Alu.cidade as cidade_alu,
    Mat.cod as cod_mat,
    Mat.data as data_mat,
    Mat.valor as valor_mat
from Alunos as Alu
right join Matriculas Mat
on Alu.cod = Mat.cod_aluno
where Mat.data <= '2023-10-31'
order by Alu.cod,Mat.cod;
